name: Cross-border Intel CI/CD

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  build-and-test:
    runs-on: ubuntu-latest
    steps:
    - uses: actions/checkout@v3
    
    - name: Set up Python
      uses: actions/setup-python@v4
      with:
        python-version: '3.10'
        
    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r backend/requirements.txt
        
    - name: Run Linting/Tests (Placeholder)
      run: |
        echo "Running static analysis..."
        # Add pytest or flake8 here in the future
        
    - name: Build Docker Image
      run: |
        docker build -t cross-border-intel-backend ./backend
        docker build -t cross-border-intel-frontend ./frontend (if custom Dockerfile exists)
        # For now we use the docker-compose to verify build
        docker-compose build

  # Deployment job - requires secrets (SSH_PRIVATE_KEY, SERVER_IP, etc.)
  deploy:
    needs: build-and-test
    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/main'
    steps:
    - name: Deploy to Server via SSH
      run: |
